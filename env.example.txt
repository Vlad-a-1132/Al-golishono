# SMTP настройки для отправки email
# Скопируйте этот файл в .env.local и заполните реальными данными

# SMTP сервер (для Yandex: smtp.yandex.ru, для Gmail: smtp.gmail.com)
SMTP_HOST=smtp.yandex.ru

# Порт SMTP (465 для SSL, 587 для TLS)
SMTP_PORT=465

# Email отправителя (должен совпадать с SMTP_USER)
SMTP_USER=zakaz@altamed-c.ru

# Пароль от почты или пароль приложения (для Yandex нужен пароль приложения)
SMTP_PASSWORD=ваш_пароль_или_пароль_приложения
